-- ------------------------------------------------------------
-- ELIMINAR BASE ANTERIOR
-- ------------------------------------------------------------

DROP DATABASE IF EXISTS bandacol;


-- ------------------------------------------------------------
-- CREAR BASE DE DATOS
-- ------------------------------------------------------------

CREATE DATABASE bandacol
CHARACTER SET utf8mb4
COLLATE utf8mb4_unicode_ci;


USE bandacol;


-- ============================================================
-- TABLA: usuarios
-- ============================================================

CREATE TABLE usuarios (

    id INT UNSIGNED NOT NULL AUTO_INCREMENT,

    nombre VARCHAR(100) NOT NULL,

    apellido VARCHAR(100) NOT NULL,

    usuario VARCHAR(50) NOT NULL,

    email VARCHAR(150) NULL,

    password VARCHAR(255) NOT NULL,

    rol ENUM(
        'admin',
        'estudiante'
    ) NOT NULL DEFAULT 'estudiante',

    estado ENUM(
        'activo',
        'inactivo'
    ) NOT NULL DEFAULT 'activo',

    fecha_creacion TIMESTAMP NOT NULL
        DEFAULT CURRENT_TIMESTAMP,

    fecha_actualizacion TIMESTAMP NOT NULL
        DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP,

    PRIMARY KEY (id),

    UNIQUE KEY uk_usuarios_usuario (usuario),

    UNIQUE KEY uk_usuarios_email (email)

) ENGINE=InnoDB
  DEFAULT CHARSET=utf8mb4
  COLLATE=utf8mb4_unicode_ci;


-- ============================================================
-- TABLA: instrumentos
-- ============================================================

CREATE TABLE instrumentos (

    id INT UNSIGNED NOT NULL AUTO_INCREMENT,

    codigo VARCHAR(50) NOT NULL,

    nombre VARCHAR(100) NOT NULL,

    tipo VARCHAR(100) NOT NULL,

    marca VARCHAR(100) NULL,

    modelo VARCHAR(100) NULL,

    numero_serie VARCHAR(100) NULL,

    estado ENUM(
        'Disponible',
        'Prestado',
        'Mantenimiento',
        'Baja'
    ) NOT NULL DEFAULT 'Disponible',

    ubicacion VARCHAR(150) NULL,

    observaciones TEXT NULL,

    fecha_registro TIMESTAMP NOT NULL
        DEFAULT CURRENT_TIMESTAMP,

    fecha_actualizacion TIMESTAMP NOT NULL
        DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP,

    PRIMARY KEY (id),

    UNIQUE KEY uk_instrumentos_codigo (codigo),

    UNIQUE KEY uk_instrumentos_serie (numero_serie),

    KEY idx_instrumentos_estado (estado)

) ENGINE=InnoDB
  DEFAULT CHARSET=utf8mb4
  COLLATE=utf8mb4_unicode_ci;


-- ============================================================
-- TABLA: prestamos
-- ============================================================

CREATE TABLE prestamos (

    id INT UNSIGNED NOT NULL AUTO_INCREMENT,

    instrumento_id INT UNSIGNED NOT NULL,

    estudiante_id INT UNSIGNED NOT NULL,

    fecha_prestamo DATETIME NOT NULL
        DEFAULT CURRENT_TIMESTAMP,

    fecha_devolucion_prevista DATE NOT NULL,

    fecha_devolucion_real DATETIME NULL,

    estado ENUM(
        'Activo',
        'Devuelto',
        'Vencido'
    ) NOT NULL DEFAULT 'Activo',

    observaciones TEXT NULL,

    PRIMARY KEY (id),

    KEY idx_prestamos_instrumento (instrumento_id),

    KEY idx_prestamos_estudiante (estudiante_id),

    KEY idx_prestamos_estado (estado),

    CONSTRAINT fk_prestamos_instrumento
        FOREIGN KEY (instrumento_id)
        REFERENCES instrumentos(id)
        ON UPDATE CASCADE
        ON DELETE RESTRICT,

    CONSTRAINT fk_prestamos_estudiante
        FOREIGN KEY (estudiante_id)
        REFERENCES usuarios(id)
        ON UPDATE CASCADE
        ON DELETE RESTRICT

) ENGINE=InnoDB
  DEFAULT CHARSET=utf8mb4
  COLLATE=utf8mb4_unicode_ci;
